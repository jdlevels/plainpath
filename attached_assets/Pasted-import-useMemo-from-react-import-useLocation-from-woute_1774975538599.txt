import { useMemo } from "react";
import { useLocation } from "wouter";

const PLAN_DETAILS: Record<string, { title: string; price: string }> = {
  starter: { title: "Starter", price: "$4.99/month" },
  pro: { title: "Pro", price: "$14.99/month" }
};

export default function Subscribe() {
  const [location] = useLocation();

  const plan = useMemo(() => {
    const url = new URL(window.location.origin + location);
    const value = url.searchParams.get("plan") ?? "starter";
    return PLAN_DETAILS[value] ? value : "starter";
  }, [location]);

  const selected = PLAN_DETAILS[plan];

  return (
    <div className="mx-auto max-w-3xl px-4 py-20 sm:px-6 lg:px-8">
      <div className="rounded-3xl border border-slate-200 bg-white p-8 shadow-sm dark:border-slate-800 dark:bg-slate-950">
        <p className="text-sm font-medium text-blue-600 dark:text-blue-400">
          Subscription setup
        </p>

        <h1 className="mt-3 text-3xl font-bold text-slate-900 dark:text-white">
          {selected.title} Plan
        </h1>

        <p className="mt-2 text-lg text-slate-600 dark:text-slate-300">
          {selected.price}
        </p>

        <p className="mt-6 text-slate-600 dark:text-slate-300">
          Billing is not connected yet. This page is a placeholder for Stripe or
          your final checkout flow.
        </p>

        <div className="mt-8 flex flex-col gap-3 sm:flex-row">
          <a
            href="mailto:hello@plainpath.app?subject=PlainPath%20Subscription%20Setup"
            className="inline-flex items-center justify-center rounded-xl bg-blue-600 px-4 py-3 text-sm font-semibold text-white hover:bg-blue-700"
          >
            Contact to Subscribe
          </a>

          <a
            href="/#pricing"
            className="inline-flex items-center justify-center rounded-xl bg-slate-100 px-4 py-3 text-sm font-semibold text-slate-900 hover:bg-slate-200 dark:bg-slate-900 dark:text-white dark:hover:bg-slate-800"
          >
            Back to Pricing
          </a>
        </div>
      </div>
    </div>
  );
}
